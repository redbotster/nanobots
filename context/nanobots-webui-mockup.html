apiVersion: nanobots.dev/v1alpha1
kind: Nanoswarm
metadata:
  name: daily-email-recap
spec:
  defaults:
    model: { provider: anthropic, name: claude-sonnet-4-6 }
    guardrails: { pii: redact, injection_threshold: 0.7, daily_budget_usd: 5, approval_required_for: [email.send] }
  trigger: { type: cron, expr: "0 7 * * 1-5", timezone: America/Chicago }
  bots:
    - id: recap
      use: recap-emails-to-pdf@0.3.0
    - id: mailer
      use: email-drive-file@1.1.0
      inputs: { to: "{{vars.notify_to}}" }
  snaps:
    - { from: recap.drive_file_id, to: mailer.file_id }
    - { from: recap.recap_json.headline, to: mailer.body_intro }
  deploy: { target: local }   # or: 1claw | kubernetes | apple
